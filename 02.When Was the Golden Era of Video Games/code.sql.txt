-- best_selling_games

SELECT *
FROM game_sales
ORDER BY games_sold DESC
LIMIT 10;

-- critics_top_ten_years

SELECT
		gs.year,
		COUNT(*) AS num_games,
		ROUND(AVG(critic_score),2) AS avg_critic_score
FROM reviews AS r
LEFT JOIN game_sales AS gs
ON r.name = gs.name
GROUP BY year
HAVING count(*) > 4
ORDER BY avg_critic_score DESC

LIMIT 10;

-- golden_years

SELECT u.year,
	   u.num_games,
	   c.avg_critic_score,
	   u.avg_user_score,
	   (c.avg_critic_score - u.avg_user_score) AS diff
FROM users_avg_year_rating AS u
LEFT JOIN critics_avg_year_rating AS c
ON u.year = c.year
	
WHERE c.avg_critic_score > 9.0 OR u.avg_user_score > 9.0
ORDER BY year;
